// bot-social.mjs — Automação IG 4:5 & FB com Banco Anti-Repetição (Dos 600 ao 1000)
// Instale: npm install node-cron
import fs from 'node:fs/promises';
import cron from 'node-cron';

const DB_FILE = './publicados-db.json';

async function carregarHistorico() {
  try { return JSON.parse(await fs.readFile(DB_FILE, 'utf8')); }
  catch { return {}; }
}

async function registrarPublicado(postKey, meta) {
  const db = await carregarHistorico();
  db[postKey] = { ...meta, publishedAt: new Date().toISOString() };
  await fs.writeFile(DB_FILE, JSON.stringify(db, null, 2));
}

export async function publicarNoInstagram({ igUserId, accessToken, imageUrl, caption }) {
  const baseUrl = `https://graph.facebook.com/v21.0/${igUserId}`;
  const mediaRes = await fetch(`${baseUrl}/media`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ image_url: imageUrl, caption, access_token: accessToken })
  });
  const mediaData = await mediaRes.json();
  if (!mediaData.id) throw new Error('Erro IG /media: ' + JSON.stringify(mediaData));

  const pubRes = await fetch(`${baseUrl}/media_publish`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ creation_id: mediaData.id, access_token: accessToken })
  });
  return await pubRes.json();
}

export async function publicarNoFacebook({ pageId, accessToken, imageUrl, caption }) {
  const res = await fetch(`https://graph.facebook.com/v21.0/${pageId}/photos`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ url: imageUrl, message: caption, access_token: accessToken })
  });
  return await res.json();
}

// Cron 12:00 e 19:00 com checagem automática contra repetição!
export function iniciarCronSocial(buscarProximoIneditoFn) {
  cron.schedule('0 12,19 * * *', async () => {
    const db = await carregarHistorico();
    const post = await buscarProximoIneditoFn(db);
    if (!post) return console.log('[CRON] Todos os posts do ciclo já foram publicados.');
    if (db[post.postKey]) return console.log('[ANTI-REPETIÇÃO] Ignorado:', post.title);

    if (post.platforms.includes('IG')) await publicarNoInstagram(post);
    if (post.platforms.includes('FB')) await publicarNoFacebook(post);
    await registrarPublicado(post.postKey, { title: post.title, platforms: post.platforms });
    console.log('[CRON] Publicado e salvo no banco:', post.title);
  }, { timezone: 'America/Sao_Paulo' });
}
